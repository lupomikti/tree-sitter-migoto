[CommandListSetShapeKey]
; Check the last set value for this $component_id and $shapekey_id...
local $value_uid = $component_id * $shapekeyed_comp_max_batches * $\MI\cfg_sk_batch_size + $shapekey_id
if $shapekey_value != $Pool_ShapeKeys_LastSetValues[$value_uid]
    ; ... and submit new value to GPU only if it is different.
    ; Calling `ShapeKeySetter` just writes new value to buffer, geometry update is deferred.
    $Pool_ShapeKeys_LastSetValues[$value_uid] = $shapekey_value
    $\MI\shapekey_id = $shapekey_id
    $\MI\shapekey_value = $shapekey_value
    Resource\MI\Output_ShapeKeyValues = ref Pool_ShapeKeyValuesRW[$component_id]
    run = CustomShader\MI\ShapeKeySetter
    ; Signal `CommandList_ShapeKeys_Apply` geometry updater that new value was submitted in this frame.
    $Pool_ShapeKeys_Component_UpdateID[$component_id] = FRAME_NUMBER
endif
